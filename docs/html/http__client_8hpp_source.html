3// L0 传输层 + 完整 curl 封装（纯协程，零线程）：
4//
5// a) 传输原语（LLM 引擎 / 流式用）：async_http_post / async_http_get / HttpStreamReader。
6// b) 通用请求原语：HttpRequest（curl 能力全映射：方法 / body / multipart / 流式上传 /
7// auth / TLS / 重定向 / 代理 / cookie）+ async_http_request / sync_http_request。
8// c) 便捷层（工具友好，非 2xx → Error）：http_get_json / http_post_json / http_post_form /
9// http_upload_file。
10// d) HttpClient（状态化，协程）：easy 句柄池连接复用（keep-alive）+ cookie jar。
11//
12// 传输后端：libcurl multi_socket 模式嫁接 asio io_context——零线程桥、单事件循环，
13// socket 由 asio 创建（CURLOPT_OPENSOCKETFUNCTION）并以 async_wait 驱动
14// curl_multi_socket_action。C API 全部封在 .cpp 的 detail 实现内。
15// TLS：Windows 走 schannel（系统证书库）；其他平台 OpenSSL。
16//
17// L0 域无关硬约束：本层不认识 StreamEvent/ChatResponse 等 chat 类型，
18// 引擎协程拿「原始 body 字节」，自行喂 SseParser → 推域事件。
19//
20// 纯协程不变式（HttpClient 同样适用）：所有 curl 回调只在绑定 executor 上执行；
21// HttpClient 非线程安全，须由单一 io_context（或 strand）驱动。

102class HttpResponse {
103public:
104 int status = 0;
105 std::string body;
106 std::vector<std::pair<std::string, std::string>> headers;
107 std::string effective_url;
108 double total_time_seconds = 0;
109
111 std::string_view header(std::string_view name) const
112 {
113 auto lower_eq = [](std::string_view a, std::string_view b) {
114 if (a.size() != b.size()) return false;
115 for (std::size_t i = 0; i < a.size(); ++i) {
116 char x = a[i], y = b[i];
117 if (x >= 'A' && x <= 'Z') x = static_cast<char>(x - 'A' + 'a');
118 if (y >= 'A' && y <= 'Z') y = static_cast<char>(y - 'A' + 'a');
119 if (x != y) return false;
120 }
121 return true;
122 };
123 for (auto const& [header_name, value] : headers)
124 if (lower_eq(header_name, name))
125 return value;
126 return {};
127 }
128
130 bool ok() const { return status / 100 == 2; }
131
133 Result<nlohmann::json> json() const;
134};
