4//
5// DefaultBehaviors（agent.hpp）的 should_compact / compact 默认成员内部调用这些函数；
6// override 时可选复用。摘要请求构造是「前缀缓存」的关键（对齐 opencode）：
7// - 摘要指令必须作**尾部 user 消息** append 进 messages，**绝不放进 system prompt**——
8// system prompt 一旦变，prompt 前缀全变 → 缓存全废（pi 把摘要指令塞
9// SUMMARIZATION_SYSTEM_PROMPT 是败笔）。
10// - 摘要请求 StreamOptions：max_tokens = min(model.max_output_tokens, 4096)、
11// thinking 关闭、temperature 0、Context.tools 留空——摘要响应出现 ToolCall
12// → 压缩失败（CompactionFailed），不做任何工具处理。
13//
14// 已知简化（诚实记录，见 PLAN_3.md 五.5.4）：
15// - find_cut_point 轮内切点是「整条消息」保留而非字符级切片：超预算单条整体保留
16// （可能略超 keep_recent_tokens），避免切断 ToolCall/ToolResult 配对。

91inline bool is_compaction_summary(Message const& message)
92{
93 if (message.role != Role::User)
94 return false;
95 for (auto const& block : message.content)
96 if (auto text = std::get_if<Text>(&block))
97 return text->text.starts_with(kCompactionSummaryPrefix);
98 return false;
99}

101inline std::string extract_summary_text(Message const& message)
102{
103 if (message.role != Role::User)
104 return {};
105 for (auto const& block : message.content)
106 if (auto text = std::get_if<Text>(&block))

108 return text->text.substr(kCompactionSummaryPrefix.size());

112inline int estimate_tokens_default(Message const& message)
113{
114 int tokens = 0;
115 for (auto const& block : message.content) {
116 tokens += std::visit([](auto const& b) -> int {
117 using T = std::decay_t<decltype(b)>;
118 if constexpr (std::is_same_v<T, Text>)
119 return static_cast<int>(b.text.size() / 4);
120 else if constexpr (std::is_same_v<T, Thinking>)
121 return static_cast<int>(b.text.size() / 4);
122 else if constexpr (std::is_same_v<T, Image>)
123 return 4800;
124 else if constexpr (std::is_same_v<T, ToolCall>) {
125 int n = static_cast<int>(b.name.size() / 4);
126 std::string const& args = b.arguments.is_string()
127 ? b.arguments.template get_ref<std::string const&>()
128 : b.arguments.dump();
129 return n + static_cast<int>(args.size() / 4);
130 }
131 else if constexpr (std::is_same_v<T, ToolResult>)
132 return static_cast<int>((b.output.size() + 8) / 4);
133 else
134 return 0;
135 }, block);
136 }
137 return tokens;
138}

140inline int estimate_context_tokens(std::vector<Message> const& messages)
141{
142 // 找最近一条带真实 token 计数的 assistant 消息（usage 锚点）：
143 // 锚点 = 该请求的真实 input+output（含此前全部上下文），其后消息逐条估算。
144 std::optional<int> anchor;
145 std::size_t anchor_tail = 0;
146 for (std::size_t i = 0; i < messages.size(); ++i) {
147 Message const& m = messages[i];
148 if (m.role == Role::Assistant && m.input_tokens() && m.output_tokens()) {
149 anchor = static_cast<int>(*m.input_tokens() + *m.output_tokens());
150 anchor_tail = i + 1;
151 }
152 }
153 int total = anchor.value_or(0);
154 for (std::size_t i = anchor_tail; i < messages.size(); ++i)
155 total += estimate_tokens_default(messages[i]);
156 return total; // 无锚点 → 全量估算（anchor_tail = 0，total 从 0 累加）
157}

164inline std::size_t find_cut_point(std::vector<Message> const& messages, CompactionSettings const& settings)
165{
166 if (messages.empty())
167 return 0;
168
169 // 1) 切 turn：每个 User 消息 = turn 起点
170 struct Turn {
171 std::size_t begin;
172 std::size_t end;
173 int tokens;
174 };
175 std::vector<Turn> turns;
176 std::size_t i = 0;
177 while (i < messages.size()) {
178 if (messages[i].role != Role::User) {
179 ++i;
180 continue;
181 }
182 std::size_t begin = i;
183 ++i;
184 while (i < messages.size() && messages[i].role != Role::User)
185 ++i;
186 int tokens = 0;
187 for (std::size_t k = begin; k < i; ++k)
188 tokens += estimate_tokens_default(messages[k]);
189 turns.push_back({ begin, i, tokens });
190 }
191 if (turns.empty())
192 return 0; // 无 user 消息 → 不压
193
194 // 2) 候选保留区 = 最后 tail_turns 个 turn
195 std::size_t tail = settings.tail_turns > 0 ? static_cast<std::size_t>(settings.tail_turns) : 1;
196 std::size_t first_candidate = turns.size() > tail ? turns.size() - tail : 0;
197
198 // 3) 从后往前累加 turn token：<= keep_recent_tokens 整轮保留，否则轮内切
199 std::size_t cut = turns[first_candidate].begin; // 兜底：整个候选区保留
200 int budget = settings.keep_recent_tokens;
201 for (std::size_t ti = turns.size(); ti > first_candidate; --ti) {
202 Turn const& turn = turns[ti - 1];
203 if (turn.tokens <= budget) {
204 cut = turn.begin;
205 budget -= turn.tokens;
206 continue;
207 }
208 // 该 turn 超出剩余预算 → 轮内从后往前保留消息；超预算单条整体保留
209 std::size_t keep_from = turn.end;
210 int remain = budget;
211 for (std::size_t k = turn.end; k > turn.begin; --k) {
212 int message_tokens = estimate_tokens_default(messages[k - 1]);
213 if (message_tokens > remain) {
214 keep_from = (remain > 0) ? (k - 1) : k; // 有预算 → 该条整体保留；无预算 → 整条进摘要
215 break;
216 }
217 remain -= message_tokens;
218 keep_from = k - 1;
219 }
220 cut = keep_from;
221 break;
222 }
223 return cut;
224}

226inline std::string build_summary_instruction(std::string_view previous_summary)
227{
228 std::string instruction =
229 "Please compress the conversation above into a concise summary, preserving:\n"
230 "- The user's goals and completed items\n"
231 "- Key decisions and conclusions\n"
232 "- Pending / unfinished items\n"
233 "- Important facts and context\n";
234 if (!previous_summary.empty()) {
235 instruction += "\nThis is the previous summary; merge and update it on top of it "
236 "(do not repeat already-covered content):\n";
237 instruction += previous_summary;
238 instruction += "\n";
239 }
240 return instruction;

243inline std::optional<SummaryRequest> build_summary_request(
244 std::vector<Message> const& messages, ModelView const& model,
245 CompactionSettings const& settings, std::string_view system_prompt,
246 std::string_view previous_summary)
247{
248 if (!agent::should_compact(estimate_context_tokens(messages), model.context_window, settings))
249 return std::nullopt;
250 std::size_t cut = find_cut_point(messages, settings);
251 if (cut == 0 || cut >= messages.size())
252 return std::nullopt; // 无可压（全对话都在保留预算内）或无保留内容
253
254 SummaryRequest request;
255 request.cut = cut;
256 request.ctx.system_prompt = std::string(system_prompt); // system 不变
257 request.ctx.messages.assign(messages.begin(), messages.begin() + static_cast<std::ptrdiff_t>(cut));
258 request.ctx.messages.push_back(
259 Message{ Role::User, { Text{ build_summary_instruction(previous_summary) } } });
260 // ctx.tools 默认空 = 留空
261 int max_out = model.max_output_tokens > 0 ? model.max_output_tokens : 4096;
262 request.opts.max_tokens = std::min(max_out, 4096);
263 request.opts.reasoning = ThinkingLevel::Off;
264 request.opts.temperature = 0;
265 return request;
266}

268inline Result<std::string> validate_summary_response(ChatResponse const& response)
269{
270 for (auto const& block : response.content)
271 if (std::get_if<ToolCall>(&block))
272 return std::unexpected(Error{ Errc::CompactionFailed, "压缩摘要响应出现工具调用" });
273 std::string summary;
274 for (auto const& block : response.content)
275 if (auto text = std::get_if<Text>(&block))
276 summary += text->text;
277 if (summary.empty())
278 return std::unexpected(Error{ Errc::CompactionFailed, "压缩摘要为空" });
279 return summary;
280}

282inline std::vector<Message> apply_summary(std::vector<Message> const& messages, std::size_t cut,
283 std::string const& summary)
284{
285 std::vector<Message> result;
286 result.reserve(1 + (messages.size() - cut));
287 result.push_back(
288 Message{ Role::User, { Text{ std::string(kCompactionSummaryPrefix) + summary } } });
289 result.insert(result.end(), messages.begin() + static_cast<std::ptrdiff_t>(cut), messages.end());
290 return result;
291}
