3// 高层 Agent 封装（对齐原版 pi packages/agent 全部 agent 功能，零裁剪）：
4// Agent<Provider, Behaviors> —— 引擎循环 + 可覆写行为（钩子 + 压缩策略一体）。
5// DefaultBehaviors —— 默认钩子 + 压缩策略；override 用「非虚名字隐藏」编译期决议，零 vtable。
6//
7// 双层循环（对齐 pi 〇.4）：
8// 内层 turn 循环：steering 注入 → 流式收集回复 → 提取工具调用 → 执行（before/after 钩子）
9// → 结果回传 → turn_end → prepareNextTurn → shouldStop → 取 steering，直到无工具调用；
10// 外层：follow-up 队列非空 → 继续；否则 agent_end。
11// 压缩：**run 内防爆**（每轮发请求前 should_compact / compact，超限才压）+ **外部主动压缩**
12// （agent.should_compact() / agent.compact()），共用同一可覆写策略。
13//
14// ⚠️ 覆盖约束：override 的钩子必须是模板方法（template<typename Provider>），签名与默认一致，
15// 不一致会「隐藏同名但签名不同」静默编译错。不用 CRTP——Agent 持有具体类型，名字隐藏已保证编译期决议。
16// ⚠️ Agent 不可拷贝 / 不可移动：async 协程（run_async / compact_async / wait_for_idle）引用内部
17// provider_ / behaviors_，Agent 生命周期必须覆盖调用期间。
18// ⚠️ 并发：不支持并发 run（一次一个 run 在途）；abort() 可从其他线程调用（最佳努力，见 cancelled()）。
19

197 // ── 运行 ──
200 std::generator<AgentEvent> run(std::vector<Message> user_messages,
201 StreamOptions const& opts = {})
202 {
203 detail::SyncStreamBridge<AgentEvent> bridge;
204 aborted_.store(false); // 每次 run 重置取消标志
205 cancel_.emplace(); // 每次 run 新建取消信号
206 // ⚠️ loop 不绑取消槽（cancellation_signal 单槽，槽留给生产协程独占）；
207 // 取消靠 aborted_ 标志检查 + 生产协程关流（llm.close()）触发 receive 返回。
208 asio::co_spawn(bridge.io(),
209 loop_async(std::move(user_messages), bridge.sink(), opts),
210 asio::detached);
211 while (std::optional<AgentEvent> event = bridge.pop_after_io())
212 co_yield std::move(*event);
213 }
214
216 asio::awaitable<void> run_async(std::vector<Message> user_messages,
217 AsyncStream<AgentEvent> sink,
218 StreamOptions const& opts = {})
219 {
220 auto ex = co_await asio::this_coro::executor;
221 aborted_.store(false);
222 cancel_.emplace();
223 auto [ec] = co_await asio::co_spawn(ex,
224 loop_async(std::move(user_messages), std::move(sink), opts),
225 asio::as_tuple(asio::use_awaitable));
226 (void)ec; // 取消以 Aborted 形式由 loop 内 fail_run 处理
227 }

232 return run({}, opts);
233 }

371 // ── 暴露信息（供外部判断要不要压 + 成本监控）──
373 Usage context_usage() const
374 {
375 Usage usage;
376 int total = estimate_context_tokens(messages_);
377 usage.total_tokens = total;
378 usage.input_tokens = total; // 上下文整体按输入计（重新发送成本近似）
379 if (last_usage_)
380 usage.output_tokens = last_usage_->output_tokens;
381 usage.cost = static_cast<double>(total) * (model_.price_input / 1e6)
382 + (last_usage_
383 ? static_cast<double>(last_usage_->output_tokens) * (model_.price_output / 1e6)
384 : 0.0);
385 return usage;
386 }

411 static double compute_cost(Usage const& usage, ModelView const& model)
412 {
413 return (static_cast<double>(usage.input_tokens) * model.price_input
414 + static_cast<double>(usage.output_tokens) * model.price_output
415 + static_cast<double>(usage.cache_read_tokens) * model.price_cache_read
416 + static_cast<double>(usage.cache_write_tokens) * model.price_cache_write) / 1e6;
417 }

568 asio::awaitable<std::vector<ToolResult>> execute_tools(std::vector<ToolCall> const& calls,
569 AsyncStream<AgentEvent>& sink)
570 {
571 struct Planned {
572 ToolCall call;
573 ToolResult result;
574 bool rejected;
575 };
576 std::vector<Planned> planned;
577 planned.reserve(calls.size());
578 for (ToolCall const& tc : calls) {
579 pending_tools_.push_back(tc.id);
580 co_await emit(sink, AgentEvent{ ToolExecStart{ tc.id, tc.name, tc.arguments } });
581 Result<bool> before = behaviors_.before_tool_call(provider_, tc, build_snapshot());
582 ToolResult result;
583 result.tool_call_id = tc.id;
584 bool rejected = false;
585 // 执行门控：不在 tools_ 允许集的工具直接拒绝（须 set_tools 显式指定），
586 // 不真执行，把「未开放」原因回传给模型让它调整。
587 if (!is_tool_allowed(tc.name)) {
588 rejected = true;
589 result.is_error = true;
590 result.output = "工具「" + tc.name + "」未开放（须 set_tools 显式指定）";
591 } else if (!before) {
592 rejected = true;
593 result.is_error = true;
594 result.output = before.error().message;
595 } else if (!*before) {
596 rejected = true;
597 result.is_error = true;
598 result.output = "工具被拒绝执行: " + tc.name;
599 }
600 planned.push_back({ tc, std::move(result), rejected });
601 }
602
603 // 每个工具的有效执行模式：per-tool（!= Default）覆盖 Agent 全局默认。
604 // 执行：Parallel 用 std::async 并发（先全部启动，后台跑），Sequential 按序内联，
605 // 结果按 planned 原顺序收集（每个工具结果带自己的 tool_call_id）。
606 std::vector<ToolExecutionMode> modes;
607 modes.reserve(planned.size());
608 for (auto const& p : planned)
609 modes.push_back(effective_tool_mode(p.call.name));
610
611 std::vector<std::future<Result<std::string>>> futures(planned.size());
612 for (std::size_t i = 0; i < planned.size(); ++i)
613 if (!planned[i].rejected && modes[i] == ToolExecutionMode::Parallel)
614 futures[i] = std::async(std::launch::async, [&p = planned[i]]() {
615 return Tools::exec(p.call.name, p.call.arguments);
616 });
617
618 for (std::size_t i = 0; i < planned.size(); ++i) {
619 Planned& p = planned[i];
620 if (p.rejected)
621 continue;
622 Result<std::string> r = (modes[i] == ToolExecutionMode::Parallel)
623 ? futures[i].get()
624 : Tools::exec(p.call.name, p.call.arguments);
625 apply_exec_result(p.result, p.call, r);
626 Result<ToolResult> after = behaviors_.after_tool_call(provider_, p.call, p.result);
627 if (after)
628 p.result = *after;
629 else {
630 p.result.is_error = true;
631 p.result.output = after.error().message;
632 }
633 }
634
635 std::vector<ToolResult> results;
636 results.reserve(planned.size());
637 for (Planned& p : planned) {
638 co_await emit(sink, AgentEvent{ ToolExecEnd{ p.call.id, p.call.name, p.result } });
639 results.push_back(std::move(p.result));
640 pending_tools_.pop_back();
641 }
642 for (ToolResult const& tr : results) {
643 Message tool_message{ Role::ToolResult, { tr } };
644 messages_.push_back(tool_message);
645 co_await emit(sink, AgentEvent{ MessageStart{ tool_message } });
646 co_await emit(sink, AgentEvent{ MessageEnd{ tool_message } });
647 }
648 co_return results;
649 }

651 // ── 核心循环（同步 run / 异步 run_async 共用）──
652 asio::awaitable<void> loop_async(std::vector<Message> inject,
653 AsyncStream<AgentEvent> sink,
654 StreamOptions opts)
655 {
656 auto ex = co_await asio::this_coro::executor;
657 streaming_.store(true);
658 last_error_.reset();
659
660 // 任何退出路径（正常 / fail_run / 取消 unwind）都关闭事件通道——
661 // 否则 SyncStreamBridge 的 pop_after_io 永远等不到 channel_closed 而挂起。
662 struct SinkGuard {
663 AsyncStream<AgentEvent>& sink;
664 ~SinkGuard() { sink.close(); }
665 } sink_guard{ sink };
666
667 // AgentStart 最先（对齐 pi agent_start = run 开始），随后才是 user 消息事件
668 if (!co_await emit(sink, AgentEvent{ AgentStart{} }))
669 co_return; // 消费端已关闭（generator break 场景）
670 if (cancelled()) {
671 co_await fail_run(sink, Error{ Errc::Aborted, "run 被取消" });
672 co_return;
673 }
674
675 // 注入初始 user 消息
676 for (Message const& m : inject) {
677 messages_.push_back(m);
678 co_await emit(sink, AgentEvent{ MessageStart{ m } });
679 co_await emit(sink, AgentEvent{ MessageEnd{ m } });
680 }
681
682 // run_start 钩子：改写初始消息 + 系统提示
683 Result<std::optional<RunStartPatch>> patch = behaviors_.run_start(provider_, messages_, system_prompt_);
684 if (!patch) {
685 co_await fail_run(sink, patch.error());
686 co_return;
687 }
688 if (*patch) {
689 messages_ = std::move((**patch).messages);
690 system_prompt_ = std::move((**patch).system_prompt);
691 }
692
693 while (true) { // 外层：follow-up
694 while (true) { // 内层 turn 循环
695 if (cancelled()) {
696 co_await fail_run(sink, Error{ Errc::Aborted, "run 被取消" });
697 co_return;
698 }
699
700 // run 内防爆：每轮发请求前自动压缩（超限才压，内建底线）
701 {
702 ContextSnapshot snap = build_snapshot();
703 if (behaviors_.should_compact(snap)) {
704 Result<std::optional<std::vector<Message>>> cr = behaviors_.compact(provider_, snap);
705 if (!cr) {
706 co_await fail_run(sink, cr.error());
707 co_return;
708 }
709 if (*cr) {
710 messages_ = std::move(**cr);
711 update_previous_summary();
712 }
713 }
714 }
715
716 co_await emit(sink, AgentEvent{ TurnStart{} });
717 co_await inject_steering(sink);
718 if (cancelled()) {
719 co_await fail_run(sink, Error{ Errc::Aborted, "run 被取消" });
720 co_return;
721 }
722
723 // 构建请求上下文
724 Context ctx;
725 ctx.system_prompt = system_prompt_;
726 ctx.messages = messages_;
727 ctx.tools = tools_;
728 Result<std::vector<Message>> tc = behaviors_.transform_context(provider_, build_snapshot());
729 if (!tc) {
730 co_await fail_run(sink, tc.error());
731 co_return;
732 }
733 ctx.messages = std::move(*tc);
734
735 StreamOptions effective = opts;
736 if (!effective.reasoning && reasoning_)
737 effective.reasoning = reasoning_;
738 Result<std::optional<StreamOptions>> br = behaviors_.before_request(provider_, effective);
739 if (!br) {
740 co_await fail_run(sink, br.error());
741 co_return;
742 }
743 if (*br)
744 effective = **br;
745 Result<std::optional<std::string>> key = behaviors_.get_api_key(provider_name_);
746 if (!key) {
747 co_await fail_run(sink, key.error());
748 co_return;
749 }
750 if (*key)
751 effective.api_key = **key;
752
753 // before_payload：引擎生成请求体 → 钩子改写 → 塞回 prebuilt_body 交还引擎。
754 // 钩子 nullopt（不改）→ 也用构建原样，避免引擎二次 build_params。
755 Result<nlohmann::json> built = provider_.build_params(model_, ctx, effective);
756 if (!built) {
757 co_await fail_run(sink, built.error());
758 co_return;
759 }
760 Result<std::optional<nlohmann::json>> payload = behaviors_.before_payload(provider_, *built);
761 if (!payload) {
762 co_await fail_run(sink, payload.error());
763 co_return;
764 }
765 effective.prebuilt_body = *payload ? **payload : *built;
766
767 // LLM 流：生产协程绑定取消信号（abort() → 干净中断）。
768 // ⚠️ lambda 按值捕获 ctx / effective：生产协程可能晚于 loop 局部析构
769 //（取消时两者独立 unwind），引用捕获会悬空。
770 // ⚠️ RAII CloseGuard：llm.close() 在正常/取消/异常任何退出路径都执行——
771 // 消费端 receive 必得 channel_closed，不会挂起。析构清理不算异常用法。
772 AsyncStream<StreamEvent> llm(ex);
773 asio::co_spawn(ex,
774 [this, ctx, effective, llm]() mutable -> asio::awaitable<void> {
775 struct CloseGuard {
776 AsyncStream<StreamEvent>& llm;
777 ~CloseGuard() { llm.close(); }
778 } guard{ llm };
779 co_await provider_.stream_async(model_, ctx, effective, llm);
780 },
781 asio::bind_cancellation_slot(cancel_->slot(), asio::detached));
782
783 // 消费流 → assistant 消息 + 工具调用
784 Message assistant{ Role::Assistant, {} };
785 bool assistant_started = false;
786 std::vector<ToolCall> tool_calls;
787 std::optional<Error> stream_error;
788 bool got_done = false;
789 while (true) {
790 Result<StreamEvent> ev = co_await llm.receive();
791 if (!ev)
792 break; // 生产协程结束（channel 关闭）
793 switch (ev->type()) {
794 case StreamEvent::Type::TextDelta: {
795 std::string text = std::get<TextDelta>(ev->data).text;
796 if (!assistant_started) {
797 assistant_started = true;
798 co_await emit(sink, AgentEvent{ MessageStart{ assistant } });
799 }
800 append_text(assistant, text);
801 co_await emit(sink, AgentEvent{ MessageUpdate{ assistant, TextDelta{ text } } });
802 break;
803 }
804 case StreamEvent::Type::ThinkingDelta: {
805 std::string text = std::get<ThinkingDelta>(ev->data).text;
806 if (!assistant_started) {
807 assistant_started = true;
808 co_await emit(sink, AgentEvent{ MessageStart{ assistant } });
809 }
810 append_thinking(assistant, text);
811 co_await emit(sink, AgentEvent{ MessageUpdate{ assistant, ThinkingDelta{ text } } });
812 break;
813 }
814 case StreamEvent::Type::ToolCallDelta: {
815 co_await emit(sink, AgentEvent{ std::get<ToolCallDelta>(ev->data) });
816 break;
817 }
818 case StreamEvent::Type::ToolCallEnd: {
819 ToolCallEnd const& ce = std::get<ToolCallEnd>(ev->data);
820 ToolCall call{ ce.id, ce.name, ce.arguments };
821 tool_calls.push_back(call);
822 assistant.push_back(ContentBlock{ call });
823 co_await emit(sink, AgentEvent{ ce });
824 break;
825 }
826 case StreamEvent::Type::Usage: {
827 break; // Done 的 usage 为准
828 }
829 case StreamEvent::Type::Done: {
830 ChatResponse& response = std::get<DoneEvent>(ev->data).response;
831 assistant = Message{ Role::Assistant, response.content };
832 record_usage(response.usage);
833 assistant.set_input_tokens(static_cast<std::size_t>(response.usage.input_tokens));
834 assistant.set_output_tokens(static_cast<std::size_t>(response.usage.output_tokens));
835 if (!assistant_started) {
836 assistant_started = true;
837 co_await emit(sink, AgentEvent{ MessageStart{ assistant } });
838 }
839 co_await emit(sink, AgentEvent{ MessageEnd{ assistant } });
840 got_done = true;
841 break;
842 }
843 case StreamEvent::Type::Error: {
844 stream_error = std::get<Error>(ev->data);
845 break;
846 }
847 }
848 if (got_done || stream_error)
849 break;
850 }
851
852 if (stream_error) {
853 co_await fail_run(sink, *stream_error);
854 co_return;
855 }
856 if (!got_done) {
857 // 流中断（channel 关闭而无终结事件）：取消 → Aborted；否则异常终断
858 if (cancelled())
859 co_await fail_run(sink, Error{ Errc::Aborted, "run 被取消" });
860 else
861 co_await fail_run(sink, Error{ Errc::NetworkError, "stream ended without terminal event" });
862 co_return;
863 }
864
865 Message assistant_copy = assistant;
866 messages_.push_back(std::move(assistant));
867
868 // 工具执行（回传 ToolResult 消息）
869 std::vector<ToolResult> turn_results;
870 if (!tool_calls.empty())
871 turn_results = co_await execute_tools(tool_calls, sink);
872 if (cancelled()) {
873 co_await fail_run(sink, Error{ Errc::Aborted, "run 被取消" });
874 co_return;
875 }
876
877 // TurnEnd + 每轮结束钩子
878 co_await emit(sink, AgentEvent{ TurnEnd{ assistant_copy, turn_results } });
879 Result<bool> stop = behaviors_.should_stop(provider_, assistant_copy);
880 if (!stop) {
881 co_await fail_run(sink, stop.error());
882 co_return;
883 }
884 Result<std::optional<NextTurnUpdate>> next = behaviors_.prepare_next_turn(
885 provider_, assistant_copy, turn_results);
886 if (!next) {
887 co_await fail_run(sink, next.error());
888 co_return;
889 }
890 if (*next) {
891 if ((**next).model) model_ = *((**next).model);
892 if ((**next).reasoning) reasoning_ = *((**next).reasoning);
893 if ((**next).context) messages_ = std::move(*((**next).context));
894 }
895
896 if (*stop)
897 break; // 提前停（应 stop）
898 if (!has_queued_steer() && tool_calls.empty())
899 break; // 无 steer 且无工具往返 → 内层结束
900 // 否则继续内层（有 steer 插队或工具往返待续）
901 }
902
903 if (!has_queued_follow_up())
904 break; // 外层结束
905 co_await inject_follow_up(sink); // 将停时注入 → 新的一轮
906 }
907
908 co_await emit(sink, AgentEvent{ AgentEnd{ messages_ } });
909 streaming_.store(false);
910 }

913 Provider provider_;
914 ModelView model_;
915 std::string system_prompt_;
916 std::vector<Message> messages_;
917 std::vector<std::string> tools_;
918 std::optional<Usage> last_usage_;
919 double cumulative_cost_ = 0;
920 std::string previous_summary_;
921 std::optional<ThinkingLevel> reasoning_;
922 std::optional<asio::cancellation_signal> cancel_;
923 std::atomic<bool> aborted_ = false;
924 Behaviors behaviors_;
