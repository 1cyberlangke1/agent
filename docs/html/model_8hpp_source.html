152 {
153 std::shared_ptr<detail::RegistrySnapshot> snap = load_snapshot();
154 for (std::string_view id : snap->static_order) {
155 if (snap->dynamic_ids.contains(id))
156 continue;
157 auto it = snap->index.find(id);
158 if (it != snap->index.end())
159 f(it->second);
160 }
161 for (std::string_view id : snap->dynamic_order)
162 f(snap->index.at(id));
163 }
