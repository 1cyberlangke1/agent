4// StreamFacade<Engine> —— 流式域四接口壳（chat / 未来图像 partial / TTS 共用）

7// 关键泛化：壳对引擎的**参数包完全透明**（完美转发），只依赖两个关联类型

28template<typename E>
29concept StreamEngine = requires(typename E::event_type const& ev) {
30 typename E::event_type;
31 typename E::result_type;
32 { E::as_done(ev) } -> std::same_as<std::optional<typename E::result_type>>;
33 { E::as_error(ev) } -> std::same_as<std::optional<Error>>;
34 requires std::constructible_from<typename E::event_type, Error>; // 兜底合成 Error 事件
35};

47template<typename Event>
48class SyncStreamBridge {
49public:
50 SyncStreamBridge()
51 : stream_(io_.get_executor(), 64)
52 {
53 }
54
55 SyncStreamBridge(SyncStreamBridge const&) = delete;
56 SyncStreamBridge& operator=(SyncStreamBridge const&) = delete;
57
59 AsyncStream<Event>& sink() { return stream_; }
60
61 asio::io_context& io() { return io_; }
62
64 asio::cancellation_signal& cancel_signal() { return signal_; }
65
67 std::optional<Event> pop_after_io()
68 {
69 while (true) {
70 std::optional<Result<Event>> item = stream_.try_receive();
71 if (item) {
72 if (*item)
73 return std::move(**item);
74 return std::nullopt; // channel closed
75 }
76 if (io_.stopped())
77 return std::nullopt;
78 io_.run_one(); // 推进网络直到有事件或流结束
79 }
80 }
81
82 ~SyncStreamBridge()
83 {
84 if (io_.stopped())
85 return;
86 signal_.emit(asio::cancellation_type::all);
87 stream_.close(); // 让生产协程的 send 失败返回
88 io_.run(); // 跑到生产协程退出，回收全部挂起回调
89 }
90
91private:
92 asio::io_context io_;
93 AsyncStream<Event> stream_;
94 asio::cancellation_signal signal_;
95};

151template<StreamEngine Engine>
152template<typename... Args>
153std::generator<typename Engine::event_type> StreamFacade<Engine>::stream(Args const&... args)
154{
155 using Event = typename Engine::event_type;
156 SyncStreamBridge<Event> bridge;
157 // 复制 sink（共享 channel）：生产协程持副本，bridge 保留自己的副本消费
158 asio::co_spawn(bridge.io(),
159 engine_.stream_async(args..., bridge.sink()),
160 asio::bind_cancellation_slot(bridge.cancel_signal().slot(),
161 asio::detached));
162
163 bool emitted_terminal = false;
164 while (std::optional<Event> event = bridge.pop_after_io()) {
165 if (Engine::as_done(*event).has_value() || Engine::as_error(*event).has_value())
166 emitted_terminal = true;
167 co_yield std::move(*event);
168 }
169 // 终结契约：channel 关闭而 Done/Error 皆缺 → 壳兜底合成 Error，绝不静默成功。
170 if (!emitted_terminal)
171 co_yield Event{ Error{ Errc::NetworkError, "stream ended without terminal event" } };
172}

174template<StreamEngine Engine>
175template<typename... Args>
176Result<typename Engine::result_type> StreamFacade<Engine>::complete(Args const&... args) const
177{
178 using ResultType = typename Engine::result_type;
179 asio::io_context io;
180 std::optional<Result<ResultType>> outcome;
181 asio::co_spawn(io, complete_async(args...),
182 [&outcome](std::exception_ptr, Result<ResultType> result) {
183 outcome.emplace(std::move(result));
184 });
185 io.run();
186 // co_spawn 完成回调必然执行（io.run() 跑完全部工作），此处 outcome 非空。
187 return std::move(*outcome);
188}

190template<StreamEngine Engine>
191template<typename... Args>
192asio::awaitable<Result<typename Engine::result_type>> StreamFacade<Engine>::complete_async(
193 Args const&... args) const
194{
195 using Event = typename Engine::event_type;
196 using ResultType = typename Engine::result_type;
197
198 asio::any_io_executor executor = co_await asio::this_coro::executor;
199 AsyncStream<Event> sink(executor);
200 // 生产协程引用 args...（栈上 const&，complete_async 返回前仍有效）；
201 // 复制 sink（共享 channel）：生产协程持副本，本协程用原对象 receive。
202 asio::co_spawn(executor, [&]() -> asio::awaitable<void> {
203 co_await engine_.stream_async(args..., sink);
204 }, asio::detached);
205
206 std::optional<ResultType> result;
207 std::optional<Error> error;
208 while (true) {
209 Result<Event> received = co_await sink.receive();
210 if (!received)
211 break; // channel closed = 生产协程已结束
212 if (std::optional<ResultType> done = Engine::as_done(*received)) {
213 result = std::move(done);
214 break;
215 }
216 if (std::optional<Error> err = Engine::as_error(*received)) {
217 error = std::move(err);
218 break;
219 }
220 }
221 // 终结契约：生产协程结束而 Done/Error 皆缺 → 合成错误，绝不静默成功。
222 if (result.has_value())
223 co_return std::move(*result);
224 if (error.has_value())
225 co_return Result<ResultType>{ std::unexpect, std::move(*error) };
226 co_return Result<ResultType>{ std::unexpect,
227 Error{ Errc::NetworkError, "stream ended without terminal event" } };
228}
