116inline void validate_args(
117 nlohmann::json const& value,
118 nlohmann::json const& schema,
119 std::string const& path,
120 std::vector<std::string>& errors)
121{
122 if (!schema.is_object())
123 return;
124
125 // ── type 检查（支持 "string" 与 ["string","null"] 两种形式）── //
126 auto type_it = schema.find("type");
127 if (type_it != schema.end()) {
128 bool matched = false;
129 bool checkable = false;
130 auto match_one = [&](nlohmann::json const& t) {
131 if (!t.is_string())
132 return;
133 checkable = true;
134 std::string_view type_name = t.template get_ref<std::string const&>();
135 if (type_name == "null") matched = matched || value.is_null();
136 else if (type_name == "object") matched = matched || value.is_object();
137 else if (type_name == "array") matched = matched || value.is_array();
138 else if (type_name == "string") matched = matched || value.is_string();
139 else if (type_name == "boolean") matched = matched || value.is_boolean();
140 else if (type_name == "number") matched = matched || value.is_number();
141 else if (type_name == "integer") matched = matched || is_integer_value(value);
142 };
143 if (type_it->is_string()) {
144 match_one(*type_it);
145 } else if (type_it->is_array()) {
146 for (auto const& t : *type_it)
147 match_one(t);
148 }
149 if (checkable && !matched) {
150 errors.push_back(path.empty()
151 ? std::string("arguments type mismatch")
152 : path + " type mismatch");
153 return; // 类型不对，后续约束无意义，避免级联误报
154 }
155 if (value.is_null())
156 return; // null 已被 ["x","null"] 接受，无更多约束可查
157 }
158
159 // ── object：required 存在性 + properties 递归 ── //
160 if (value.is_object()) {
161 auto required_it = schema.find("required");
162 if (required_it != schema.end() && required_it->is_array()) {
163 for (auto const& r : *required_it) {
164 if (!r.is_string())
165 continue;
166 std::string_view key = r.template get_ref<std::string const&>();
167 if (!value.contains(key))
168 errors.push_back(path + "." + std::string(key) + " is required");
169 }
170 }
171 auto properties_it = schema.find("properties");
172 if (properties_it != schema.end() && properties_it->is_object()) {
173 for (auto it = value.begin(); it != value.end(); ++it) {
174 auto prop_it = properties_it->find(it.key());
175 if (prop_it != properties_it->end())
176 validate_args(it.value(), *prop_it, path + "." + it.key(), errors);
177 }
178 }
179 }
180
181 // ── array：长度约束 + items 递归 ── //
182 if (value.is_array()) {
183 auto take_count = [&](char const* key) -> std::int64_t {
184 auto it = schema.find(key);
185 if (it != schema.end() && it->is_number_integer())
186 return it->template get<std::int64_t>();
187 return -1; // 缺失或畸形 → 不检查
188 };
189 if (std::int64_t min_items = take_count("minItems");
190 min_items >= 0 && std::cmp_less(value.size(), min_items))
191 errors.push_back(path + " too few items");
192 if (std::int64_t max_items = take_count("maxItems");
193 max_items >= 0 && std::cmp_greater(value.size(), max_items))
194 errors.push_back(path + " too many items");
195
196 auto items_it = schema.find("items");
197 if (items_it != schema.end() && items_it->is_object()) {
198 for (std::size_t i = 0; i < value.size(); ++i)
199 validate_args(value[i], *items_it,
200 path + "[" + std::to_string(i) + "]", errors);
201 }
202 }
203
204 // ── enum：值必须在枚举列表内 ── //
205 auto enum_it = schema.find("enum");
206 if (enum_it != schema.end() && enum_it->is_array()) {
207 bool found = false;
208 for (auto const& e : *enum_it) {
209 if (e == value) {
210 found = true;
211 break;
212 }
213 }
214 if (!found)
215 errors.push_back(path + " not in enum");
216 }
217}
