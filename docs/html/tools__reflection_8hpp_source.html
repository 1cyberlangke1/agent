80template<typename T>
81consteval FieldType type_to_field_type()
82{
83 if constexpr (std::is_same_v<T, bool>) return FieldType::Boolean;
84 else if constexpr (std::is_same_v<T, std::string>) return FieldType::String;
85 else if constexpr (std::is_integral_v<T>) return FieldType::Integer;
86 else if constexpr (std::is_floating_point_v<T>) return FieldType::Number;
87 else if constexpr (std::is_enum_v<T>) return FieldType::Enum;
88 else if constexpr (is_specialization_c<T, std::vector>) return FieldType::Array;
89 else if constexpr (is_std_array<T>::value) return FieldType::Array;
90 else if constexpr (is_specialization_c<T, std::optional>) return type_to_field_type<typename T::value_type>();
91 else if constexpr (std::is_class_v<T>) return FieldType::Object;
92 else static_assert(sizeof(T) == 0,
93 "type not supported as tool parameter: use bool, std::string, integral, "
94 "floating point, enum, vector<T>, array<T,N>, optional<T>, or a struct");
95}

147template<typename Outer, typename Field>
148consteval bool directly_recursive()
149{

184template<typename T>
185nlohmann::json build_type_schema()
186{
187 // 全剥 optional：optional<optional<X>> 只包一层 null，语义与 JSON 一致
188 using BareT = strip_optional_all_t<T>;
189 constexpr FieldType field_type = type_to_field_type<T>();
190 nlohmann::json schema;
191
192 if constexpr (field_type == FieldType::Object) {
193 schema = build_schema<BareT>();
194 }
195 else if constexpr (field_type == FieldType::Array) {
196 schema["type"] = "array";
197 schema["items"] = build_type_schema<typename BareT::value_type>();
198 if constexpr (is_std_array<BareT>::value) {
199 schema["minItems"] = std::tuple_size_v<BareT>;
200 schema["maxItems"] = std::tuple_size_v<BareT>;
201 }
202 }
203 else if constexpr (field_type == FieldType::Enum) {
204 schema["type"] = "string";
205 schema["enum"] = nlohmann::json::array();
206 template for (constexpr auto e : std::define_static_array(
207 std::meta::enumerators_of(^^BareT)))
208 {
209 schema["enum"].push_back(std::string(std::meta::identifier_of(e)));
210 }
211 }
212 else {
213 schema["type"] = field_type_to_json(field_type);
214 }
215
216 // optional 包装：type 从 "xxx" 变为 ["xxx", "null"]
217 if constexpr (is_specialization_c<T, std::optional>) {
218 schema["type"] = nlohmann::json::array(
219 {schema["type"].template get<std::string>(), "null"});
220 }
221
222 return schema;
223}

229template<typename T>
230nlohmann::json build_schema()
231{
232 nlohmann::json schema = nlohmann::json::object();
233 schema["type"] = "object";
234 schema["properties"] = nlohmann::json::object();
235 schema["required"] = nlohmann::json::array();
236
237 template for (constexpr auto m : reflect_members<T>())
238 {
239 using RawType = std::decay_t<decltype(std::declval<T>().[:m:])>;
240 static_assert(!directly_recursive<T, RawType>(),
241 "recursive type detected: struct cannot contain itself "
242 "(e.g. struct Node { std::vector<Node> children; })");
243
244 auto field_name = std::meta::identifier_of(m);
245 nlohmann::json prop = build_type_schema<RawType>();
246
247 if constexpr (find_desc(m) != nullptr)
248 prop["description"] = find_desc(m);
249
250 if constexpr (std::meta::has_default_member_initializer(m)) {
251 if constexpr (std::is_trivially_default_constructible_v<RawType>) {
252 // T() 在常量求值中构造（string 等成员走 transient 分配），
253 // 提取的字段值本身必须是编译期常量（标量/枚举满足）
254 constexpr auto default_value = T().[:m:];
255 if constexpr (std::is_same_v<RawType, bool>)
256 prop["default"] = default_value;
257 else if constexpr (std::is_integral_v<RawType>)
258 prop["default"] = default_value;
259 else if constexpr (std::is_floating_point_v<RawType>)
260 prop["default"] = default_value;
261 else if constexpr (std::is_enum_v<RawType>) {
262 // 枚举默认值以枚举名字符串写入，与 enum 约束的取值一致
263 if constexpr (enum_default_name<RawType>(default_value) != nullptr)
264 prop["default"] = enum_default_name<RawType>(default_value);
265 }
266 } else if constexpr (std::is_same_v<RawType, std::string>) {
267 // string 默认值非编译期常量（堆分配），运行时构造一次提取
268 T defaults{};
269 prop["default"] = defaults.[:m:];
270 }
271 }
272
273 schema["properties"][std::string(field_name)] = std::move(prop);
274
275 // 无默认值且非 optional 的字段才是 required
276 if constexpr (!std::meta::has_default_member_initializer(m)
277 && !is_specialization_c<RawType, std::optional>)
278 {
279 schema["required"].push_back(std::string(field_name));
280 }
281 }
282
283 return schema;
284}

311template<typename T>
312Result<T> extract_json(nlohmann::json const& j, std::string const& path)
313{
314 if constexpr (is_specialization_c<T, std::optional>) {
315 using InnerT = typename T::value_type;
316 if (j.is_null())
317 return T(std::nullopt);
318 Result<InnerT> inner = extract_json<InnerT>(j, path);
319 if (!inner)
320 return std::unexpected(inner.error());
321 return T(std::move(*inner));
322 }
323 else if constexpr (is_specialization_c<T, std::vector>) {
324 using ElemT = typename T::value_type;
325 if (!j.is_array())
326 return std::unexpected(Error{Errc::InvalidArgs, path + " expects array"});
327 T result;
328 result.reserve(j.size());
329 for (std::size_t i = 0; i < j.size(); ++i) {
330 Result<ElemT> elem = extract_json<ElemT>(
331 j[i], path + "[" + std::to_string(i) + "]");
332 if (!elem)
333 return std::unexpected(elem.error());
334 result.push_back(std::move(*elem));
335 }
336 return result;
337 }
338 else if constexpr (is_std_array<T>::value) {
339 using ElemT = typename T::value_type;
340 constexpr std::size_t expected_size = std::tuple_size_v<T>;
341 if (!j.is_array() || j.size() != expected_size)
342 return std::unexpected(Error{Errc::InvalidArgs,
343 path + " expected " + std::to_string(expected_size) + " elements, got "
344 + (j.is_array() ? std::to_string(j.size()) : std::string("non-array"))});
345 T result{};
346 for (std::size_t i = 0; i < expected_size; ++i) {
347 Result<ElemT> elem = extract_json<ElemT>(
348 j[i], path + "[" + std::to_string(i) + "]");
349 if (!elem)
350 return std::unexpected(elem.error());
351 result[i] = std::move(*elem);
352 }
353 return result;
354 }
355 else if constexpr (std::is_enum_v<T>) {
356 if (!j.is_string())
357 return std::unexpected(Error{Errc::InvalidArgs,
358 path + " expects string for enum"});
359 std::string const& str = j.template get_ref<std::string const&>();
360 bool found = false;
361 T value{};
362 template for (constexpr auto e : std::define_static_array(
363 std::meta::enumerators_of(^^T)))
364 {
365 if (std::meta::identifier_of(e) == str) {
366 value = std::meta::extract<T>(e);
367 found = true;
368 }
369 }
370 if (!found)
371 return std::unexpected(Error{Errc::InvalidArgs,
372 path + " unknown enum value: " + str});
373 return value;
374 }
375 else if constexpr (std::is_same_v<T, bool>) {
376 if (!j.is_boolean())
377 return std::unexpected(Error{Errc::InvalidArgs, path + " expects boolean"});
378 return j.template get<bool>();
379 }
380 else if constexpr (std::is_integral_v<T>) {
381 if (j.is_number_unsigned()) {
382 std::uint64_t raw = j.template get<std::uint64_t>();
383 if (!std::in_range<T>(raw))
384 return std::unexpected(Error{Errc::InvalidArgs, path + " out of range"});
385 return static_cast<T>(raw);
386 }
387 if (j.is_number_integer()) {
388 std::int64_t raw = j.template get<std::int64_t>();
389 if (!std::in_range<T>(raw))
390 return std::unexpected(Error{Errc::InvalidArgs, path + " out of range"});
391 return static_cast<T>(raw);
392 }
393 if (j.is_number_float()) {
394 double value = j.template get<double>();
395 if (std::trunc(value) == value && std::abs(value) <= 9007199254740992.0) {
396 std::int64_t raw = static_cast<std::int64_t>(value);
397 if (!std::in_range<T>(raw))
398 return std::unexpected(Error{Errc::InvalidArgs, path + " out of range"});
399 return static_cast<T>(raw);
400 }
401 }
402 return std::unexpected(Error{Errc::InvalidArgs, path + " expects integer"});
403 }
404 else if constexpr (std::is_floating_point_v<T>) {
405 if (!j.is_number())
406 return std::unexpected(Error{Errc::InvalidArgs, path + " expects number"});
407 return j.template get<T>();
408 }
409 else if constexpr (std::is_same_v<T, std::string>) {
410 if (!j.is_string())
411 return std::unexpected(Error{Errc::InvalidArgs, path + " expects string"});
412 return j.template get<std::string>();
413 }
414 else if constexpr (std::is_class_v<T>) {
415 return assign_from_json<T>(j);
416 }
417 else {
418 static_assert(sizeof(T) == 0, "unsupported type in extract_json");
419 }
420}

430template<typename T>
431 requires std::is_class_v<T>
432Result<T> assign_from_json(nlohmann::json const& j)
433{
434 if (!j.is_object())
435 return std::unexpected(Error{Errc::InvalidArgs,
436 "arguments must be a JSON object"});
437
438 T obj{};
439
440 template for (constexpr auto m : reflect_members<T>())
441 {
442 using RawType = std::decay_t<decltype(std::declval<T>().[:m:])>;
443 auto name = std::meta::identifier_of(m);
444 std::string path = "." + std::string(name);
445
446 auto it = j.find(name);
447 if (it == j.end()) {
448 if constexpr (!std::meta::has_default_member_initializer(m)
449 && !is_specialization_c<RawType, std::optional>)
450 {
451 return std::unexpected(Error{Errc::InvalidArgs,
452 path + " is required"});
453 }
454 continue;
455 }
456
457 if (it->is_null() && !is_specialization_c<RawType, std::optional>)
458 return std::unexpected(Error{Errc::InvalidArgs,
459 path + " cannot be null (not optional)"});
460
461 Result<RawType> value = extract_json<RawType>(*it, path);
462 if (!value)
463 return std::unexpected(value.error());
464 obj.[:m:] = std::move(*value);
465 }
466
467 return obj;
468}

551template<typename T>
552ToolBase<T>::Registrar::Registrar()
553{
554 using Params = typename T::params_type;
555 constexpr detail::ToolMeta meta = detail::reflect_tool_meta<T>();
556
557 ToolInfo info{
558 std::string(meta.name),
559 meta.description != nullptr ? std::string(meta.description) : std::string(),
560 detail::schema_of<Params>()
561 };
562
563 // ArgsCheck::Tool：assign_from_json 已全量校验，跳过 exec 的 schema 预校验。
564 // 工具类若定义 `static asio::awaitable<Result<std::string>> invoke_async(params_type const&)`
565 // → 异步注册（exec 内部桥接）；否则用同步 invoke（两种工具并存）。
566 // 执行模式：工具类若声明 `static constexpr ToolExecutionMode execution_mode` → per-tool 生效。
567 constexpr ToolExecutionMode mode = detail::tool_execution_mode<T>();
568 Result<void> registered;
569 if constexpr (requires(Params const& p) {
570 { T::invoke_async(p) } -> std::same_as<asio::awaitable<Result<std::string>>>;
571 }) {
572 registered = Tools::reg(std::move(info),
573 [](nlohmann::json args) -> asio::awaitable<Result<std::string>> {
574 Result<Params> params = detail::assign_from_json<Params>(args);
575 if (!params)
576 co_return std::unexpected(params.error());
577 co_return co_await T::invoke_async(std::move(*params));
578 },
579 ArgsCheck::Tool, mode);
580 } else {
581 registered = Tools::reg(std::move(info),
582 [](nlohmann::json args) -> Result<std::string> {
583 Result<Params> params = detail::assign_from_json<Params>(args);
584 if (!params)
585 return std::unexpected(params.error());
586 return T::invoke(std::move(*params));
587 },
588 ArgsCheck::Tool, mode);
589 }
590
591 // 静态初始化阶段无处返回错误，stderr 警告是唯一可靠通道
592 // （重复注册几乎必然是程序员错误，宜早暴露）
593 if (!registered)
594 std::fprintf(stderr, "[agent] tool '%s' registration failed: %s\n",
595 meta.name, registered.error().message.c_str());
596}
